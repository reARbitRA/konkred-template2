    // خروجی: API routes
    //         Prisma schema + migrations
    //         Validation (Zod)
    //         Error handling
    //         Auth middleware

// مرحله ۳: INTEGRATE + VERIFY (یکپارچه‌سازی)
const finalProduct = await generate({
  prompt: userPromptTemplates.integrator,
  context: { architecture, frontend, backend },
  model: 'claude-fable-5', // بهترین برای review
  // خروجی: Integration code
  //         Type consistency check
  //         Import/export alignment
  //         Final project structure

// ← اینجاست که ۱۰۰۰ prompt تخصصی تو
